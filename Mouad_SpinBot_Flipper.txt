local CONFIG_FILE = "MouadSpinConfig.json"

-- Load saved config
local function loadConfig()
    if isfile and isfile(CONFIG_FILE) then
        local ok, data = pcall(function()
            return HttpService:JSONDecode(readfile(CONFIG_FILE))
        end)
        if ok and data then
            Config.Name = data.Name or Config.Name
            Config.Color = data.Color or Config.Color
        end
    end
end

local function saveConfig()
    if writefile then
        pcall(function()
            writefile(CONFIG_FILE, HttpService:JSONEncode(Config))
        end)
    end
end

loadConfig()

local function getAccent()
    return Color3.fromRGB(Config.Color[1], Config.Color[2], Config.Color[3])
end

local function refresh()
    Character = LocalPlayer.Character
    if not Character then return end
    Humanoid = Character:FindFirstChildOfClass("Humanoid")
    Root = Character:FindFirstChild("HumanoidRootPart")
end

LocalPlayer.CharacterAdded:Connect(function()
    task.wait(0.25)
    refresh()
end)
refresh()

    isCompact = state
    if isCompact then
        TweenService:Create(Main, TweenInfo.new(0.18), {
            Size = UDim2.fromOffset(200, 130)
        }):Play()
        SpeedLabel.Visible = false
        SpeedFrame.Visible = false
        NameLabel.Visible = false
        NameBox.Visible = false
        ColorLabel.Visible = false
        ColorFrame.Visible = false
        SaveBtn.Visible = false
        SpinBtn.Position = UDim2.fromOffset(10, 38)
        FlipBtn.Position = UDim2.fromOffset(10, 78)
    else
        TweenService:Create(Main, TweenInfo.new(0.18), {
            Size = UDim2.fromOffset(250, 320)
        }):Play()
        SpeedLabel.Visible = true
        SpeedFrame.Visible = true
        NameLabel.Visible = true
        NameBox.Visible = true
        ColorLabel.Visible = true
        ColorFrame.Visible = true
        SaveBtn.Visible = true
        SpinBtn.Position = UDim2.fromOffset(10, 40)
        FlipBtn.Position = UDim2.fromOffset(10, 82)
    end
end

UserInputService.InputBegan:Connect(function(input, gpe)
    if gpe then return end
    if input.KeyCode == Enum.KeyCode.RightControl then
        setSpin(not SpinOn)
    elseif input.KeyCode == Enum.KeyCode.RightAlt then
        setFlip(not FlipOn)
    end
end)

-- ========================= HARD ANTI-AIM =========================
RunService.Heartbeat:Connect(function(dt)
    refresh()
    if not Root or not Humanoid or Humanoid.Health <= 0 then return end
    if not (SpinOn or FlipOn) then return end

    local pos = Root.Position
    local vel = Root.AssemblyLinearVelocity

    if SpinOn then
        yaw = yaw + SpinSpeed * dt * 10
    else
        yaw = yaw + 6 * dt * 8
    end

    local cf = CFrame.new(pos)

    if FlipOn then
        lastFlip = lastFlip + dt
        local interval = FlipInterval + (math.noise(tick() * 3) * 0.04)
        if lastFlip >= interval then
            lastFlip = 0
            flipState = (flipState + 1) % 12
        end

        local patterns = {
            {180, 0, 0},
            {0, 0, 180},
            {90, 90, 0},
            {-90, -140, 50},
            {150, 60, -110},
            {-175, 180, 90},
            {70, -95, 170},
            {-120, 130, -80},
            {200, 45, 60},
            {-60, 200, -150},
            {110, -70, 140},
            {-160, 100, -40},
        }

        local p = patterns[flipState + 1]
        cf = cf * CFrame.Angles(0, math.rad(yaw), 0)
        cf = cf * CFrame.Angles(math.rad(p[1]), math.rad(p[2]), math.rad(p[3]))
    else
        cf = cf * CFrame.Angles(0, math.rad(yaw), 0)
    end

    Root.CFrame = cf
    Root.AssemblyLinearVelocity = vel
end)
