local function loadConfig()
    pcall(function()
        if not (isfile and readfile) or not isfile(CONFIG_FILE) then return end
        local data = HttpService:JSONDecode(readfile(CONFIG_FILE))
        for k, v in pairs(data) do
            if CONFIG[k] ~= nil then
                if (KEYBIND_KEYS[k] or k == "BypassKeybind") and type(v) == "string" then
                    CONFIG[k] = Enum.KeyCode[v] or CONFIG[k]
                else
                    CONFIG[k] = v
                end
            end
        end
    end)
end
loadConfig()

local function saveConfig()
    pcall(function()
        if writefile then
            local toSave = {}
            for k, v in pairs(CONFIG) do
                if typeof(v) == "EnumItem" then toSave[k] = v.Name
                else toSave[k] = v end
            end
            writefile(CONFIG_FILE, HttpService:JSONEncode(toSave))
        end
    end)
end

_G._AmbitiousTPDownMethod        = CONFIG.TPDownMode or "NORMAL"
_G._AmbitiousSlowFallActive      = false
_G._AmbitiousTPDownJumpLocked    = false
_G.AmbitiousAutoTPHeight         = CONFIG.AutoTPHeight or 20

function _G._TPDownSpeedValue()
  local ch  = Player.Character
  local hum = ch and ch:FindFirstChildOfClass("Humanoid")
  return (hum and hum.WalkSpeed) or 16
end

function _G._TPDownJumpProxy()
  local ch = Player.Character
  return ch and ch:FindFirstChild("HumanoidRootPart")
end

do
  local TP_DOWN_FALLBACK_Y   = -7.00
  local TP_DOWN_MAX_DISTANCE = 4000
  local TP_DOWN_MAX_STEPS    = 12

  local EXCLUDE_FILTER = (function()
    local ok, v = pcall(function() return Enum.RaycastFilterType.Exclude end)
    if ok and v then return v end
    return nil
  end)()

  local function standOffset(hum, hrp)
    local half = 1
    pcall(function() half = hrp.Size.Y * 0.5 end)
    if half <= 0 then half = 1 end
    local off = half * 3
    pcall(function()
      if hum and hum.RigType == Enum.HumanoidRigType.R6 then
        off = half * 3
      elseif hum then
        off = half + (hum.HipHeight or (half * 2))
      end
    end)
    if off ~= off or off < 1 or off > 12 then off = half * 3 end
    return off + 0.15
  end

  local function rayDown(fromY, x, z, filter)
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = filter
    if EXCLUDE_FILTER then params.FilterType = EXCLUDE_FILTER end
    params.IgnoreWater = true
    pcall(function() params.RespectCanCollide = true end)
    local res
    pcall(function()
      res = workspace:Raycast(Vector3.new(x, fromY, z), Vector3.new(0, -TP_DOWN_MAX_DISTANCE, 0), params)
    end)
    return res
  end

  function _G.AmbitiousGetGroundTPY(char, hrp, hum)
    char = char or Player.Character
    if not char then return nil end
    hrp = hrp or char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    hum = hum or char:FindFirstChildOfClass("Humanoid")
    local filter = { char }
    pcall(function()
      for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= Player and plr.Character then table.insert(filter, plr.Character) end
      end
    end)
    local origin = hrp.Position
    local halfY  = 1
    pcall(function() halfY = hrp.Size.Y * 0.5 end)
    local startY  = origin.Y - halfY - 0.1
    local groundY = nil
    for _ = 1, TP_DOWN_MAX_STEPS do
      local res = rayDown(startY, origin.X, origin.Z, filter)
      if not res then break end
      local part = res.Instance
      local skip = false
      if part then
        pcall(function()
          if part:IsA("BasePart") and part.CanCollide == false then skip = true end
          local model = part:FindFirstAncestorOfClass("Model")
          if model and model:FindFirstChildOfClass("Humanoid") then skip = true end
        end)
      else skip = true end
      if skip then
        if part then table.insert(filter, part) end
        startY = res.Position.Y - 0.05
      else
        groundY = res.Position.Y
        break
      end
    end
    if not groundY then return nil end
    return groundY + standOffset(hum, hrp), groundY, (origin.Y - groundY)
  end

  function _G.AmbitiousResolveTPDownY(char, hrp, hum)
    local ok, y = pcall(_G.AmbitiousGetGroundTPY, char, hrp, hum)
    if ok and type(y) == "number" then return y, true end
    return TP_DOWN_FALLBACK_Y, false
  end
end

function _G.AmbitiousSlowFallDown(targetY)
  if _G._AmbitiousSlowFallActive then return end
  _G._AmbitiousSlowFallActive   = true
  _G._AmbitiousTPDownJumpLocked = true
  task.spawn(function()
    local POWER    = 115
    local deadline = os.clock() + 6
    local landed   = false
    while os.clock() < deadline do
      local char = Player.Character
      local hrp  = char and char:FindFirstChild("HumanoidRootPart")
      local hum  = char and char:FindFirstChildOfClass("Humanoid")
      if not hrp or not hum or hum.Health <= 0 then break end
      if hum.FloorMaterial ~= Enum.Material.Air then landed = true break end
      local proxy = _G._TPDownJumpProxy()
      local part  = proxy or hrp
      local v     = part.Velocity
      part.Velocity = Vector3.new(v.X, -POWER, v.Z)
      RunService.Heartbeat:Wait()
    end
    if not landed then
      local proxy = _G._TPDownJumpProxy()
      if proxy then
        local v = proxy.Velocity
        pcall(function() proxy.Velocity = Vector3.new(v.X, 0, v.Z) end)
      end
    end
    _G._AmbitiousSlowFallActive = false
    local groundDeadline = os.clock() + 8
    while os.clock() < groundDeadline do
      local c2 = Player.Character
      local h2 = c2 and c2:FindFirstChildOfClass("Humanoid")
      if not h2 or h2.Health <= 0 then break end
      if h2.FloorMaterial ~= Enum.Material.Air then break end
      RunService.Heartbeat:Wait()
    end
    _G._AmbitiousTPDownJumpLocked = false
  end)
end

function _G.AmbitiousDoTPDown(force)
  local autoTPHeight = _G.AmbitiousAutoTPHeight or 20
  local char = Player.Character
  if not char then return end
  local hrp = char:FindFirstChild("HumanoidRootPart")
  if not hrp then return end
  local hum = char:FindFirstChildOfClass("Humanoid")
  if not hum then return end
  if not force then
    if hum.FloorMaterial ~= Enum.Material.Air then return end
    if hrp.Position.Y < autoTPHeight then return end
  end
  local targetY = _G.AmbitiousResolveTPDownY(char, hrp, hum)
  if not force and (hrp.Position.Y - targetY) < 0.75 then return end
  local method = _G._AmbitiousTPDownMethod or "NORMAL"
  if method == "SLOW" then
    _G.AmbitiousSlowFallDown(targetY)
    return
  end
  hrp.CFrame = CFrame.new(hrp.Position.X, targetY, hrp.Position.Z)
             * CFrame.Angles(0, select(2, hrp.CFrame:ToEulerAnglesYXZ()), 0)
  hrp.AssemblyLinearVelocity = Vector3.zero
  _G._AmbitiousTPDownJumpLocked = true
  task.spawn(function()
    local groundDeadline = os.clock() + 8
    while os.clock() < groundDeadline do
      local c2 = Player.Character
      local h2 = c2 and c2:FindFirstChildOfClass("Humanoid")
      if not h2 or h2.Health <= 0 then break end
      if h2.FloorMaterial ~= Enum.Material.Air then break end
      RunService.Heartbeat:Wait()
    end
    _G._AmbitiousTPDownJumpLocked = false
  end)
end

function _G.AmbitiousRunTPDown(forceNormal)
  if not forceNormal then
    _G.AmbitiousDoTPDown(true)
    return
  end
  local prev = _G._AmbitiousTPDownMethod
  _G._AmbitiousTPDownMethod = "NORMAL"
  pcall(_G.AmbitiousDoTPDown, true)
  _G._AmbitiousTPDownMethod = prev
end

-- Auto TP Down
local _autoTPConn = nil
local _autoTPLastRun = 0
local function _clearAutoTP()
  if _autoTPConn then pcall(function() _autoTPConn:Disconnect() end); _autoTPConn = nil end
end
local function _startAutoTP()
  CONFIG.AutoTPEnabled = true
  saveConfig()
  _clearAutoTP()
  _autoTPLastRun = 0
  _autoTPConn = RunService.Heartbeat:Connect(function()
    if not CONFIG.AutoTPEnabled then _clearAutoTP() return end
    local now = tick()
    if now - _autoTPLastRun < 0.1 then return end
    _autoTPLastRun = now
    pcall(_G.AmbitiousDoTPDown, false)
  end)
end
local function _stopAutoTP()
  CONFIG.AutoTPEnabled = false
  saveConfig()
  _clearAutoTP()
end

local infJumpConn = nil
local lastJumpTime = 0
local HOP_POWER = 55
local HOP_COOLDOWN = 0.08

local function startInfJump()
    if infJumpConn then infJumpConn:Disconnect(); infJumpConn = nil end
    infJumpConn = RunService.Heartbeat:Connect(function()
        if not CONFIG.InfiniteJumpEnabled then return end
        pcall(function()
            local char = Player.Character
            if not char then return end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if not hum then return end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if not hrp then return end
            local now = tick()
            local wantsToJump = UserInputService:IsKeyDown(Enum.KeyCode.Space)
                or UserInputService:IsKeyDown(Enum.KeyCode.ButtonA)
                or hum.Jump == true
            if wantsToJump and (now - lastJumpTime) >= HOP_COOLDOWN then
                lastJumpTime = now
                local v = hrp.Velocity
                hrp.Velocity = Vector3.new(v.X, HOP_POWER, v.Z)
            end
        end)
    end)
end
local function stopInfJump()
    if infJumpConn then infJumpConn:Disconnect(); infJumpConn = nil end
end

RunService.Heartbeat:Connect(function()
    if CONFIG.InfiniteJumpEnabled then
        if not infJumpConn then startInfJump() end
    else
        if infJumpConn then stopInfJump() end
    end
end)

Player.CharacterAdded:Connect(function()
    if CONFIG.InfiniteJumpEnabled then
        task.wait(0.5)
        stopInfJump()
        startInfJump()
    end
end)

        if hum.PlatformStand or st == Enum.HumanoidStateType.Physics or st == Enum.HumanoidStateType.Ragdoll or st == Enum.HumanoidStateType.FallingDown then return end
        if CONFIG.AimbotV1Enabled and _G.MNL_Aimbot_Active then return end
        local md = hum.MoveDirection
        if md.Magnitude > 0 then
            local spd = getCurrentSpeed()
            local v = hrp.Velocity
            hrp.Velocity = Vector3.new(md.X * spd, v.Y, md.Z * spd)
        end
    end)
end

-- AUTO CARRY SPEED
--============================================================
local SoftState = {animals = {}, lastScan = 0}
local AS = {locked = false, wasLow = false, savedMode = nil}

local function isMyBase(plot)
    local sign = plot:FindFirstChild("PlotSign")
    local yb = sign and sign:FindFirstChild("YourBase")
    return yb and yb:IsA("BillboardGui") and yb.Enabled
end

local function scanAnimals()
    local list = {}
    local plots = workspace:FindFirstChild("Plots")
    if not plots then SoftState.animals = list; return end
    for _, plot in ipairs(plots:GetChildren()) do
        if plot:IsA("Model") and not isMyBase(plot) then
            local podiums = plot:FindFirstChild("AnimalPodiums")
            if podiums then
                for _, podium in ipairs(podiums:GetChildren()) do
                    if podium:IsA("Model") then
                        local base = podium:FindFirstChild("Base")
                        local spawn = base and base:FindFirstChild("Spawn")
                        if spawn then table.insert(list, spawn.Position) end
                    end
                end
            end
        end
    end
    SoftState.animals = list
end

local function nearestAnimalDistance()
    local char = Player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return math.huge end
    local now = tick()
    if now - SoftState.lastScan >= CONFIG.SoftScanInterval then
        SoftState.lastScan = now; pcall(scanAnimals)
    end
    local best = math.huge
    for _, pos in ipairs(SoftState.animals) do
        local d = (hrp.Position - pos).Magnitude
        if d < best then best = d end
    end
    return best
end
local function inSoftRadius() return nearestAnimalDistance() <= CONFIG.SoftRadius end

local function setSpeedMode(mode, fromAuto)
    if mode ~= "Normal" and mode ~= "Carry" and mode ~= "Lagger" and mode ~= "Lagger Carry" then mode = "Normal" end
    if AS.locked and not fromAuto and (mode ~= "Carry" and mode ~= "Lagger Carry") then return end
    CONFIG.CurrentMode = mode
    fireModeChanged()
    saveConfig()
end

local function updateAutoCarry()
    if not CONFIG.AutoCarryEnabled then return end
    local char = Player.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    local isLow = hum.WalkSpeed < 25
    if CONFIG.AutoCarryMode == "SOFT" then isLow = isLow or inSoftRadius() end
    if isLow == AS.wasLow then return end
    AS.wasLow = isLow
    if isLow then
        AS.locked = true
        if CONFIG.CurrentMode ~= "Carry" and CONFIG.CurrentMode ~= "Lagger Carry" then
            AS.savedMode = CONFIG.CurrentMode
            if CONFIG.CurrentMode == "Lagger" then setSpeedMode("Lagger Carry", true) else setSpeedMode("Carry", true) end
        end
    else
        AS.locked = false
        if AS.savedMode then
            local s = AS.savedMode; AS.savedMode = nil
            setSpeedMode(s, true)
        end
    end
    fireModeChanged()
end

task.spawn(function() while true do task.wait(0); updateAutoCarry() end end)

local function startAntiDrop()
    if AntiDrop.active then return end
    if not (getrawmetatable and setreadonly and newcclosure and checkcaller) then return end
    local mt = getrawmetatable(game)
    if not mt then return end
    AntiDrop.oldIdx = mt.__index
    AntiDrop.oldNewIdx = mt.__newindex
    setreadonly(mt, false)
    mt.__index = newcclosure(function(self, key)
        if not checkcaller() and (key == "AssemblyLinearVelocity" or key == "Velocity") and typeof(self) == "Instance" and self:IsA("BasePart") and self.Name == "HumanoidRootPart" and self:IsDescendantOf(Player.Character) then
            return AntiDrop.spoofedVelocity
        end
        return AntiDrop.oldIdx(self, key)
    end)
    mt.__newindex = newcclosure(function(self, key, value)
        if not checkcaller() and (key == "AssemblyLinearVelocity" or key == "Velocity") and typeof(self) == "Instance" and self:IsA("BasePart") and self.Name == "HumanoidRootPart" and self:IsDescendantOf(Player.Character) then
            AntiDrop.spoofedVelocity = value
            return
        end
        return AntiDrop.oldNewIdx(self, key, value)
    end)
    setreadonly(mt, true)
    AntiDrop.active = true
    task.spawn(function()
        while AntiDrop.active do
            local char = Player.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if hrp and hum and hum.Health > 0 then
                local md = hum.MoveDirection
                local fakeSpeed = 16
                if md.Magnitude > 0.05 then
                    local u = md.Unit
                    AntiDrop.spoofedVelocity = Vector3.new(u.X * fakeSpeed, hrp.Velocity.Y, u.Z * fakeSpeed)
                else
                    AntiDrop.spoofedVelocity = Vector3.new(0, hrp.Velocity.Y, 0)
                end
            end
            RunService.Stepped:Wait()
        end
    end)
end

local function stopAntiDrop()

    if AntiDrop.oldIdx then
        local mt = getrawmetatable(game)
        if mt then
            setreadonly(mt, false)
            mt.__index = AntiDrop.oldIdx
            mt.__newindex = AntiDrop.oldNewIdx
            setreadonly(mt, true)
        end
    end
    AntiDrop.active = false
end

local KatanaCache = {}
local KatanaConns = setmetatable({}, {__mode = "k"})

do
    local function setKatanaSound(tool, soundId)
        for _, sound in ipairs(tool:GetDescendants()) do
            if sound:IsA("Sound") and sound.Name == "Slash" then
                local original = sound:GetAttribute("MNL_OriginalSlash")
                if soundId then
                    if original == nil then sound:SetAttribute("MNL_OriginalSlash", sound.SoundId) end
                    sound.SoundId = soundId
                elseif original ~= nil then
                    sound.SoundId = original
                    sound:SetAttribute("MNL_OriginalSlash", nil)
                end
            end
        end
    end

    local function loadKatana(id)
        if KatanaCache[id] then return KatanaCache[id] end
        local ok, objects = pcall(function() return game:GetObjects(id) end)
        if not ok or not objects or #objects == 0 then return nil end
        KatanaCache[id] = objects[1]
        return KatanaCache[id]
    end

    local function clearMNLKatana(tool)
        local toolHandle = tool and tool:FindFirstChild("Handle")
        if not toolHandle then return end
        for _, v in ipairs(toolHandle:GetChildren()) do
            if v.Name == "MNL_KatanaModel" then v:Destroy() end
            if v:IsA("Weld") and v:GetAttribute("MNL_KatanaWeld") then v:Destroy() end
        end
    end

    local function applyKatana(tool, enabled)
        if not tool or tool.Name:lower() ~= "bat" then return end
        local toolHandle = tool:FindFirstChild("Handle")
        if not toolHandle then return end

        if not enabled then
            if tool:GetAttribute("MNL_OriginalGrip") then
                tool.Grip = tool:GetAttribute("MNL_OriginalGrip")
                tool:SetAttribute("MNL_OriginalGrip", nil)
            end
            clearMNLKatana(tool)
            toolHandle.Transparency = 0
            toolHandle.CanCollide = true
            setKatanaSound(tool, nil)
            if KatanaConns[tool] then KatanaConns[tool]:Disconnect(); KatanaConns[tool] = nil end
            return
        end

        local entry = KatanaList[CONFIG.KatanaIndex]
        if not entry then return end
        local katanaRoot = loadKatana(entry.id)
        if not katanaRoot then return end

        if tool:GetAttribute("MNL_OriginalGrip") == nil then
            tool:SetAttribute("MNL_OriginalGrip", tool.Grip)
        end

        clearMNLKatana(tool)

        toolHandle.Transparency = 1
        toolHandle.CanCollide = false
        toolHandle.Size = Vector3.new(0.3, 1, 0.3)

        setKatanaSound(tool, entry.soundId)

        local modelClone = katanaRoot:Clone()
        modelClone.Name = "MNL_KatanaModel"
        modelClone.Parent = toolHandle

        local function makeKinematic(inst)
            if inst:IsA("BasePart") then
                inst.Anchored = false
                inst.CanCollide = false
                inst.Massless = true
            end
            for _, c in ipairs(inst:GetChildren()) do makeKinematic(c) end
        end
        makeKinematic(modelClone)

        local primaryPart = modelClone.PrimaryPart
        if not primaryPart then
            for _, inst in ipairs(modelClone:GetDescendants()) do
                if inst:IsA("BasePart") then primaryPart = inst; break end
            end
        end

        if primaryPart then
            local weld = Instance.new("Weld")
            weld.Name = "MNL_KatanaWeld"
            weld:SetAttribute("MNL_KatanaWeld", true)
            weld.Part0 = toolHandle
            weld.Part1 = primaryPart
            weld.C0 = entry.c0
            weld.Parent = toolHandle
        end

        tool.GripForward = Vector3.new(0, -1, 0)
        tool.GripRight   = Vector3.new(1,  0, 0)
        tool.GripUp      = Vector3.new(0,  0, 1)
        tool.GripPos     = Vector3.new(0,  0, 0)

        if tool.Parent == Player.Character then
            task.defer(function()
                if tool.Parent == Player.Character then
                    tool.Grip = tool.Grip
                end
            end)
        end
    end

    local function refreshKatanaOnAllBats()
        local char = Player.Character
        if not char then return end
        for _, obj in ipairs(char:GetChildren()) do
            if obj:IsA("Tool") and obj.Name:lower() == "bat" then applyKatana(obj, CONFIG.KatanaEnabled) end
        end
        local bp = Player:FindFirstChild("Backpack")
        if bp then
            for _, obj in ipairs(bp:GetChildren()) do
                if obj:IsA("Tool") and obj.Name:lower() == "bat" then applyKatana(obj, CONFIG.KatanaEnabled) end
            end
        end
    end

    local function onToolEquipped(tool)
        if tool:IsA("Tool") and tool.Name:lower() == "bat" and CONFIG.KatanaEnabled then applyKatana(tool, true) end
    end

    local function onToolUnequipped(tool)
        if tool:IsA("Tool") and tool.Name:lower() == "bat" then applyKatana(tool, false) end
    end

    local function bindKatanaChar(char)
        char.ChildAdded:Connect(function(child)
            if child:IsA("Tool") then task.defer(function() onToolEquipped(child) end) end
        end)
        char.ChildRemoved:Connect(function(child)
            if child:IsA("Tool") then onToolUnequipped(child) end
        end)
        task.wait(1)
        for _, child in ipairs(char:GetChildren()) do
            if child:IsA("Tool") then onToolEquipped(child) end
        end
        local bp = Player:FindFirstChild("Backpack")
        if bp then
            bp.ChildAdded:Connect(function(child)
                if child:IsA("Tool") and child.Name:lower() == "bat" and CONFIG.KatanaEnabled then
                    task.defer(function() applyKatana(child, true) end)
                end
            end)
        end
    end

    if Player.Character then bindKatanaChar(Player.Character) end
    Player.CharacterAdded:Connect(bindKatanaChar)

    _G._MNL_refreshKatanaOnAllBats = refreshKatanaOnAllBats
end

local function refreshKatanaOnAllBats()
    if _G._MNL_refreshKatanaOnAllBats then _G._MNL_refreshKatanaOnAllBats() end

    local function updateRot(look, root) if not root or look.Magnitude < 0.01 then return end; local axis = root.CFrame.LookVector:Cross(look.Unit); local angle = math.asin(math.clamp(axis.Magnitude, -1, 1)); if axis.Magnitude > 0.01 then root.AssemblyAngularVelocity = axis.Unit * angle * 80 else root.AssemblyAngularVelocity = Vector3.zero end end
    local function getBat() local c = Player.Character; if c then local eq = c:FindFirstChildOfClass("Tool"); if eq then return eq end; for _, t in ipairs(Player.Backpack:GetChildren()) do if t:IsA("Tool") and t.Name:lower():find("bat") then return t end end; for _, t in ipairs(Player.Backpack:GetChildren()) do if t:IsA("Tool") then return t end end end; return nil end

    startAimbot = function()
        local char = Player.Character; if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid"); local root = char:FindFirstChild("HumanoidRootPart"); if not hum or not root then return end
        hum.AutoRotate = false
        local bat = getBat(); if bat and bat.Parent ~= char then hum:EquipTool(bat) end
        if not ZurichData.predictionSphere then createSphere() end
        if ZurichData.conn then ZurichData.conn:Disconnect() end
        _G.MNL_Aimbot_Active = true
        ZurichData.conn = RunService.RenderStepped:Connect(function(dt)
            if not CONFIG.AimbotV1Enabled then if ZurichData.conn then ZurichData.conn:Disconnect(); ZurichData.conn = nil end; _G.MNL_Aimbot_Active = false; return end
            ZurichData.targetPlayer = getNearestPlayer()
            if not ZurichData.targetPlayer or not ZurichData.targetPlayer.Character then if ZurichData.predictionSphere then ZurichData.predictionSphere.Transparency = 0.4 end; ZurichData.targetPlayer = nil; ZurichData.lastTargetPos = nil; ZurichData.targetVelocity = Vector3.zero; ZurichData.smoothedVelocity = Vector3.zero; ZurichData.velocityHistory = {}; ZurichData.accelerationHistory = {}; ZurichData.aerialVelocityHistory = {}; ZurichData.verticalVelocityHistory = {}; ZurichData.aerialSmoothVelocity = Vector3.zero; ZurichData.airborneTime = 0; ZurichData.highYVelocityTime = 0; ZurichData.previousDirection = nil; ZurichData.wasAirborne = false; ZurichData.lastYVelocity = 0; ZurichData.peakHeight = 0; ZurichData.isMultiJumping = false; return end
            local tr = ZurichData.targetPlayer.Character:FindFirstChild("HumanoidRootPart"); if not tr then if ZurichData.predictionSphere then ZurichData.predictionSphere.Transparency = 0.4 end; return end
            if ZurichData.predictionSphere then ZurichData.predictionSphere.Transparency = 0.4 end
            local tPos = tr.Position; local myPos = root.Position
            if ZurichData.lastTargetPos then
                local dPos = tPos - ZurichData.lastTargetPos; local rawV = dPos / dt
                rawV = clampVel(rawV, ZurichData.targetVelocity, ZCfg.MAX_VELOCITY_CHANGE)
                local hV = Vector3.new(rawV.X, 0, rawV.Z); if hV.Magnitude > ZCfg.MAX_HORIZONTAL_VELOCITY then hV = hV.Unit * ZCfg.MAX_HORIZONTAL_VELOCITY; rawV = Vector3.new(hV.X, rawV.Y, hV.Z) end
                local cA = (rawV - ZurichData.targetVelocity) / dt; table.insert(ZurichData.accelerationHistory, cA); if #ZurichData.accelerationHistory > ZurichData.MAX_ACCEL_HISTORY then table.remove(ZurichData.accelerationHistory, 1) end
                table.insert(ZurichData.verticalVelocityHistory, rawV.Y); if #ZurichData.verticalVelocityHistory > ZurichData.MAX_VERTICAL_HISTORY then table.remove(ZurichData.verticalVelocityHistory, 1) end
                ZurichData.targetVelocity = rawV; ZurichData.smoothedVelocity = ZurichData.smoothedVelocity:Lerp(ZurichData.targetVelocity, ZCfg.VELOCITY_SMOOTHING)
                table.insert(ZurichData.velocityHistory, ZurichData.targetVelocity); if #ZurichData.velocityHistory > ZurichData.MAX_HISTORY then table.remove(ZurichData.velocityHistory, 1) end
            end
            ZurichData.lastTargetPos = tPos
            if math.abs(ZurichData.targetVelocity.Y) > ZCfg.JUMPBOOST_Y_THRESHOLD then ZurichData.highYVelocityTime = ZurichData.highYVelocityTime + dt else ZurichData.highYVelocityTime = 0 end
            local isAir = checkAir(tr)
            if isAir then
                ZurichData.airborneTime = ZurichData.airborneTime + dt; if tPos.Y > ZurichData.peakHeight then ZurichData.peakHeight = tPos.Y end
                if ZurichData.airborneTime >= ZCfg.MIN_AIRBORNE_TIME then table.insert(ZurichData.aerialVelocityHistory, ZurichData.targetVelocity); if #ZurichData.aerialVelocityHistory > ZurichData.MAX_AERIAL_HISTORY then table.remove(ZurichData.aerialVelocityHistory, 1) end; ZurichData.aerialSmoothVelocity = ZurichData.aerialSmoothVelocity:Lerp(ZurichData.targetVelocity, ZCfg.AERIAL_SMOOTHING) end
                ZurichData.wasAirborne = true
            else
                ZurichData.airborneTime = 0; ZurichData.wasAirborne = false; ZurichData.aerialVelocityHistory = {}; ZurichData.aerialSmoothVelocity = Vector3.zero; ZurichData.lastGroundedPosition = tPos; ZurichData.peakHeight = 0
            end
            local infJ = isInfJump(); local floater = isFloating(); local jBoost = isJumpBoost(); local eBoost = isExtremeBoost(); local err = isErratic(); local aV = getAvg(ZurichData.velocityHistory, true); local aA = getAvg(ZurichData.accelerationHistory, true); local qTurn = detectDirChange(ZurichData.targetVelocity); local strafe = isAerialStrafing(); local trulyAir = isAir and ZurichData.airborneTime >= ZCfg.MIN_AIRBORNE_TIME; ZurichData.isMultiJumping = detectMultiJump(ZurichData.targetVelocity.Y); local fastFall = isFallingFromHeight(tPos, ZurichData.targetVelocity.Y); local aYV = getAvg(ZurichData.verticalVelocityHistory, false); ZurichData.lastYVelocity = ZurichData.targetVelocity.Y
            local pVel = ZurichData.targetVelocity; local pAcc = aA; local useCur = false
            if eBoost then useCur = true; pVel = Vector3.new(aV.X, 0, aV.Z); pAcc = Vector3.zero
            elseif jBoost then local h = Vector3.new(aV.X, 0, aV.Z); pVel = Vector3.new(h.X, ZurichData.targetVelocity.Y * 0.15, h.Z); pAcc = Vector3.new(aA.X, 0, aA.Z)
            elseif infJ or floater then local h = Vector3.new(aV.X, 0, aV.Z); pVel = Vector3.new(h.X, ZurichData.targetVelocity.Y * 0.5, h.Z); pAcc = Vector3.new(aA.X * 0.5, 0, aA.Z * 0.5)
            elseif trulyAir and strafe then local a = getAvg(ZurichData.aerialVelocityHistory, true); pVel = Vector3.new(ZurichData.aerialSmoothVelocity.X * ZCfg.AERIAL_DIRECTION_CHANGE_WEIGHT + a.X * (1 - ZCfg.AERIAL_DIRECTION_CHANGE_WEIGHT), aYV, ZurichData.aerialSmoothVelocity.Z * ZCfg.AERIAL_DIRECTION_CHANGE_WEIGHT + a.Z * (1 - ZCfg.AERIAL_DIRECTION_CHANGE_WEIGHT)); pAcc = Vector3.new(aA.X * 0.3, 0, aA.Z * 0.3)
            elseif trulyAir then pVel = Vector3.new(ZurichData.aerialSmoothVelocity.X, aYV, ZurichData.aerialSmoothVelocity.Z); pAcc = Vector3.zero
            elseif err then pVel = Vector3.new(ZurichData.smoothedVelocity.X, ZurichData.targetVelocity.Y, ZurichData.smoothedVelocity.Z); pAcc = Vector3.new(aA.X * 0.7, 0, aA.Z * 0.7) end
            local sPos = useCur and tPos or predServer(tPos, pVel, pAcc, ZurichData.currentPing, qTurn, trulyAir, strafe, fastFall, ZurichData.isMultiJumping)
            local pTime = ZCfg.PREDICTION_TIME * 1.1
            if err then pTime = pTime * 0.6 elseif qTurn then pTime = pTime * 1.2 elseif trulyAir and strafe then pTime = pTime * 0.7 elseif trulyAir and fastFall then pTime = pTime * 1.3 elseif trulyAir then pTime = pTime * 0.85 end
            local predP = trulyAir and predAerial(sPos, pVel, pTime, strafe, fastFall, ZurichData.isMultiJumping) or (sPos + pVel * pTime)
            local vOff = Vector3.zero
            if not trulyAir and not eBoost and not jBoost and not infJ then
                if ZurichData.targetVelocity.Y < ZCfg.FALLING_THRESHOLD or ZurichData.targetVelocity.Y > ZCfg.RISING_THRESHOLD then vOff = Vector3.new(0, ZurichData.targetVelocity.Y * ZCfg.VERTICAL_OFFSET_MULTIPLIER, 0) end
            end
            predP = predP + vOff
            local iOff = Vector3.zero; local hV = Vector3.new(pVel.X, 0, pVel.Z); if hV.Magnitude > 1 and not useCur then iOff = hV.Unit * ZCfg.PREDICT_AHEAD end
            local iPoint = predP + iOff
            updateSphere(iPoint, dt)
            local toT = iPoint - myPos; if toT.Magnitude > 0.1 then updateRot(toT, root) end
            local aDist = (tPos - myPos).Magnitude
            if aDist <= ZCfg.ACTIVATE_DISTANCE then
                local cTime = tick()
                if cTime - ZurichData.lastActivationTime >= 0.3 then
                    if useCur or (err and not trulyAir) then iPoint = sPos elseif trulyAir then iPoint = predAerial(sPos, pVel, ZCfg.ACTIVATION_DELAY, strafe, fastFall, ZurichData.isMultiJumping) else iPoint = sPos + pVel * ZCfg.ACTIVATION_DELAY end
                    local tool = char:FindFirstChildOfClass("Tool"); if tool then tool:Activate() end
                    ZurichData.lastActivationTime = cTime
                end
            end
            local dir = iPoint - myPos
            if dir.Magnitude > ZCfg.MIN_FOLLOW_DISTANCE then
                local cSpd = tonumber(CONFIG.AimbotV1Speed) or 55; cSpd = math.clamp(cSpd, 1, 120)
                root.AssemblyLinearVelocity = dir.Unit * cSpd
            else
                root.AssemblyLinearVelocity = Vector3.new(0, root.AssemblyLinearVelocity.Y * 0.5, 0)
            end
        end)
    end

    stopAimbot = function()
        if ZurichData.conn then ZurichData.conn:Disconnect(); ZurichData.conn = nil end
        if ZurichData.predictionSphere then ZurichData.predictionSphere:Destroy(); ZurichData.predictionSphere = nil end
        _G.MNL_Aimbot_Active = false
        local char = Player.Character
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid"); if hum then hum.AutoRotate = true end
            local root = char:FindFirstChild("HumanoidRootPart"); if root then root.AssemblyAngularVelocity = Vector3.zero end
        end
    end
end

local Steal = {
    AutoStealEnabled = false,
    StealRadius = 55,
    StealDuration = 0.1,
    Mode = "half",
    HalfFireRange = 10,
    HalfHoldMin = 1.3,
    HalfHoldMax = 2.6,
    HalfEntryDelay = 0.3,
    Data = {}
}
local isStealing = false
local stealStartTime = nil
local autoGrabConn = nil

local function isMyPlotByName(plotName)
    local plots = workspace:FindFirstChild("Plots"); if not plots then return false end
    local plot = plots:FindFirstChild(plotName); if not plot then return false end
    local sign = plot:FindFirstChild("PlotSign")
    if sign then
        local yb = sign:FindFirstChild("YourBase")
        if yb and yb:IsA("BillboardGui") then return yb.Enabled == true end
    end
    return false
end

local function findNearestPrompt()
    local char = Player.Character; if not char then return nil end
    local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
    if not root then return nil end
    local plots = workspace:FindFirstChild("Plots"); if not plots then return nil end
    local nearest, dist = nil, math.huge
    for _, plot in ipairs(plots:GetChildren()) do
        if plot:IsA("Model") and not isMyPlotByName(plot.Name) then
            local pods = plot:FindFirstChild("AnimalPodiums")
            if pods then
                for _, pod in ipairs(pods:GetChildren()) do
                    local base = pod:FindFirstChild("Base")
                    local sp = base and base:FindFirstChild("Spawn")
                    if sp then
                        local d = (sp.Position - root.Position).Magnitude
                        if d <= Steal.StealRadius and d < dist then
                            local found = nil
                            local att = sp:FindFirstChild("PromptAttachment")
                            if att then
                                for _, pr in ipairs(att:GetChildren()) do
                                    if pr:IsA("ProximityPrompt") and pr.ActionText and pr.ActionText:find("Steal") then
                                        found = pr
                                    end
                                end
                            end
                            if not found then
                                for _, pr in ipairs(sp:GetDescendants()) do
                                    if pr:IsA("ProximityPrompt") and pr.ActionText and pr.ActionText:find("Steal") then
                                        found = pr
                                    end
                                end
                            end
                            if found then nearest, dist = found, d end
                        end
                    end
                end
            end
        end
    end
    return nearest
end

local function _promptDist(prompt)
    local char = Player.Character; if not char then return math.huge end
    local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
    if not root then return math.huge end
    local part = prompt.Parent
    if part and part:IsA("Attachment") then part = part.Parent end
    if part and part:IsA("BasePart") then return (part.Position - root.Position).Magnitude end
    local ok, cf = pcall(function() return prompt.Parent and prompt.Parent.WorldPosition end)
    if ok and cf then return (cf - root.Position).Magnitude end
    return math.huge
end

local function executeSteal(prompt)
    if isStealing then return end
    if not Steal.Data[prompt] then
        Steal.Data[prompt] = {hold = {}, trigger = {}, ready = true}
        if getconnections then
            for _, c in ipairs(getconnections(prompt.PromptButtonHoldBegan)) do
                if c.Function then table.insert(Steal.Data[prompt].hold, c.Function) end
            end
            for _, c in ipairs(getconnections(prompt.Triggered)) do
                if c.Function then table.insert(Steal.Data[prompt].trigger, c.Function) end
            end
        end
    end
    local data = Steal.Data[prompt]; if not data.ready then return end
    data.ready = false; isStealing = true; stealStartTime = tick()
    if Steal.Mode == "half" then
        task.spawn(function()
            for _, fn in ipairs(data.hold) do task.spawn(fn) end
            task.wait(Steal.HalfHoldMin)
            local inRange = _promptDist(prompt) <= Steal.HalfFireRange
            while true do
                local el = tick() - stealStartTime
                if el > Steal.HalfHoldMax or not prompt.Parent then break end
                if _promptDist(prompt) <= Steal.HalfFireRange then
                    if not inRange then task.wait(Steal.HalfEntryDelay) end
                    for _, fn in ipairs(data.trigger) do task.spawn(fn) end; break
                end
                task.wait()
            end
            task.wait(0.05); data.ready = true; isStealing = false
        end)
    else
        task.spawn(function()
            for _, fn in ipairs(data.hold) do task.spawn(fn) end
            local el = 0; while el < Steal.StealDuration do el = el + task.wait() end
            for _, fn in ipairs(data.trigger) do task.spawn(fn) end
            task.wait(0.05); data.ready = true; isStealing = false
        end)
    end
end

local function startAutoGrab()
    if autoGrabConn then return end
    autoGrabConn = RunService.Heartbeat:Connect(function()
        if not Steal.AutoStealEnabled or isStealing then return end
        local p = findNearestPrompt(); if p then executeSteal(p) end
    end)
end

local function stopAutoGrab()
    if autoGrabConn then autoGrabConn:Disconnect(); autoGrabConn = nil end
    isStealing = false
end

                EnvironmentDiffuseScale = Lighting.EnvironmentDiffuseScale,
                EnvironmentSpecularScale = Lighting.EnvironmentSpecularScale,
            }
        end
        optimizerV2Effects = {}
        Lighting.GlobalShadows = false
        Lighting.FogEnd = 1e10
        Lighting.Brightness = 1
        Lighting.EnvironmentDiffuseScale = 0
        Lighting.EnvironmentSpecularScale = 0
        for _, effect in ipairs(Lighting:GetChildren()) do
            if effect:IsA("BlurEffect") or effect:IsA("SunRaysEffect")
            or effect:IsA("ColorCorrectionEffect") or effect:IsA("BloomEffect")
            or effect:IsA("DepthOfFieldEffect") then
                optimizerV2Effects[effect] = effect.Enabled
                effect.Enabled = false
            end
        end
    end

    local function restoreCrystalUltraLighting()
        local state = optimizerV2Lighting
        if state then
            pcall(function()
                Lighting.Brightness = state.Brightness
                Lighting.ClockTime = state.ClockTime
                Lighting.Ambient = state.Ambient
                Lighting.OutdoorAmbient = state.OutdoorAmbient
                Lighting.ExposureCompensation = state.ExposureCompensation
                Lighting.GlobalShadows = state.GlobalShadows
                Lighting.FogEnd = state.FogEnd
                Lighting.EnvironmentDiffuseScale = state.EnvironmentDiffuseScale
                Lighting.EnvironmentSpecularScale = state.EnvironmentSpecularScale
            end)
        end
        for effect, enabled in pairs(optimizerV2Effects) do
            if effect and effect.Parent then pcall(function() effect.Enabled = enabled end) end
        end
        optimizerV2Lighting = nil
        optimizerV2Effects = {}
        for clothing, originalParent in pairs(optimizerV2DetachedClothing) do
            if clothing and originalParent and originalParent.Parent then
                pcall(function() clothing.Parent = originalParent end)
            end
        end
        optimizerV2DetachedClothing = {}
    end

    startOptimizerV2 = function()
        if optimizerV2Active then return end
        optimizerV2Active = true
        applyCrystalUltraLighting()
        processCrystalUltraWorld()
        if optimizerV2DescConn then optimizerV2DescConn:Disconnect() end
        optimizerV2DescConn = workspace.DescendantAdded:Connect(function(obj)
            if not optimizerV2Active then return end
            applyCrystalUltraDerender(obj)
            if obj:IsA("Model") and obj:FindFirstChild("Humanoid") then
                task.defer(function()
                    if not optimizerV2Active or not obj.Parent then return end
                    for _, child in ipairs(obj:GetDescendants()) do
                        applyCrystalUltraDerender(child)
                    end
                end)
            end
        end)
    end

    stopOptimizerV2 = function()
        if not optimizerV2Active then return end
        optimizerV2Active = false
        if optimizerV2DescConn then optimizerV2DescConn:Disconnect(); optimizerV2DescConn = nil end
        restoreCrystalUltraLighting()
    end

    local StretchState = { Enabled = false, Mode = "V1", DefaultFOV = 70 }
    local stretchConnection = nil

    local function normalizeStretchCamera()
        local cam = workspace.CurrentCamera
        if not cam then return end
        local cf = cam.CFrame
        pcall(function()
            cam.CFrame = CFrame.lookAt(cf.Position, cf.Position + cf.LookVector, cf.UpVector)
        end)
    end

    local function applyStretchFrame()
        local cam = workspace.CurrentCamera
        if not cam then return end
        if StretchState.Mode == "V2" then
            cam.FieldOfView = 90
            pcall(function()
                cam.CFrame = cam.CFrame * CFrame.new(0, 0, 0, 1, 0, 0, 0, 0.7, 0, 0, 0, 1)
            end)
        else
            cam.FieldOfView = 107
        end
    end

    local function stopStretchRenderer(resetCamera)
        if stretchConnection then
            stretchConnection:Disconnect()
            stretchConnection = nil
        end
        if resetCamera then
            if StretchState.Mode == "V2" then normalizeStretchCamera() end
            local cam = workspace.CurrentCamera
            if cam then
                cam.FieldOfView = StretchState.DefaultFOV
            end
        end
    end

    local function startStretchRenderer()
        stopStretchRenderer(false)
        applyStretchFrame()
        stretchConnection = RunService.RenderStepped:Connect(function()
            if StretchState.Enabled then
                applyStretchFrame()
            end
        end)
    end

    enableStretch = function()
        if StretchState.Enabled then return end
        StretchState.Enabled = true
        startStretchRenderer()
    end

    disableStretch = function()
        if not StretchState.Enabled then return end
        StretchState.Enabled = false
        stopStretchRenderer(true)
    end

    setStretchMode = function(mode)
        if mode ~= "V1" and mode ~= "V2" then return end
        if StretchState.Mode == mode then return end
        local wasEnabled = StretchState.Enabled
        if wasEnabled then stopStretchRenderer(true) end
        StretchState.Mode = mode
        if wasEnabled then startStretchRenderer() end
    end
end

local ESPState = {
    Players  = CONFIG.ESPPlayers,
    Tracers  = CONFIG.ESPTracers,
    Skeleton = CONFIG.ESPSkeleton,
    Origin   = CONFIG.ESPOrigin or "Down",
    Accent   = PALETTE.accent,

}
local espEnabled = false
local espActivePlayers = {}

local R15_LINKS = {
    {"Head", "UpperTorso"}, {"UpperTorso", "LowerTorso"},
    {"UpperTorso", "LeftUpperArm"}, {"LeftUpperArm", "LeftLowerArm"}, {"LeftLowerArm", "LeftHand"},
    {"UpperTorso", "RightUpperArm"}, {"RightUpperArm", "RightLowerArm"}, {"RightLowerArm", "RightHand"},
    {"LowerTorso", "LeftUpperLeg"}, {"LeftUpperLeg", "LeftLowerLeg"}, {"LeftLowerLeg", "LeftFoot"},
    {"LowerTorso", "RightUpperLeg"}, {"RightUpperLeg", "RightLowerLeg"}, {"RightLowerLeg", "RightFoot"}
}
local R6_LINKS = {
    {"Head", "Torso"},
    {"Torso", "Left Arm"}, {"Torso", "Right Arm"},
    {"Torso", "Left Leg"}, {"Torso", "Right Leg"}
}

local function createESP()
    local esp = {
        Tracer    = Drawing.new("Line"),
        Highlight = Instance.new("Highlight"),
        Lines     = {},
    }
    esp.Tracer.Color        = ESPState.Accent
    esp.Tracer.Thickness    = 3
    esp.Tracer.Transparency = 1
    esp.Tracer.ZIndex       = 2
    esp.Highlight.Name                 = "ESP_Highlight"
    esp.Highlight.FillColor            = ESPState.Accent
    esp.Highlight.OutlineColor         = ESPState.Outline
    esp.Highlight.FillTransparency     = 0.4
    esp.Highlight.OutlineTransparency  = 0.1
    esp.Highlight.DepthMode            = Enum.HighlightDepthMode.AlwaysOnTop
    local parentGui
    local ok, robGui = pcall(function() return game:GetService("CoreGui"):FindFirstChild("RobloxGui") end)
    if ok and robGui then parentGui = robGui else parentGui = Player:FindFirstChildOfClass("PlayerGui") end
    if not parentGui then parentGui = game:GetService("CoreGui") end
    pcall(function() esp.Highlight.Parent = parentGui end)
    return esp
end

local function removeESP(plr)
    local data = espActivePlayers[plr]
    if not data then return end
    if data.esp.Tracer then pcall(function() data.esp.Tracer:Remove() end) end
    if data.esp.Highlight then pcall(function() data.esp.Highlight:Destroy() end) end
    for _, line in ipairs(data.esp.Lines) do
        pcall(function() line:Remove() end)
    end
    espActivePlayers[plr] = nil
end

local function updateESP(plr, esp)
    if not espEnabled then
        esp.Tracer.Visible = false
        esp.Highlight.Enabled = false
        for _, line in ipairs(esp.Lines) do line.Visible = false end
        return
    end
    local character = plr.Character
    local hum = character and character:FindFirstChildOfClass("Humanoid")
    local trackPart = character and (character:FindFirstChild("HumanoidRootPart") or character.PrimaryPart or character:FindFirstChild("UpperTorso") or character:FindFirstChild("Torso") or character:FindFirstChild("Head"))
    if character and trackPart and (not hum or hum.Health > 0) then
        esp.Highlight.Adornee = character
        esp.Highlight.Enabled = ESPState.Players
        local pos, onScreen = workspace.CurrentCamera:WorldToViewportPoint(trackPart.Position)
        if onScreen then
            if ESPState.Tracers then
                local viewportSize = workspace.CurrentCamera.ViewportSize
                if ESPState.Origin == "Body" then
                    local localRoot = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
                    local bodyPos, bodyVisible
                    if localRoot then bodyPos, bodyVisible = workspace.CurrentCamera:WorldToViewportPoint(localRoot.Position) end
                    esp.Tracer.From = bodyVisible and Vector2.new(bodyPos.X, bodyPos.Y) or Vector2.new(viewportSize.X / 2, viewportSize.Y)
                else
                    esp.Tracer.From = Vector2.new(viewportSize.X / 2, (ESPState.Origin == "Up" and 0 or viewportSize.Y))
                end
                esp.Tracer.To = Vector2.new(pos.X, pos.Y)
                esp.Tracer.Visible = true
            else
                esp.Tracer.Visible = false
            end
            if ESPState.Skeleton then
                if #esp.Lines == 0 then
                    for i = 1, 14 do
                        local line = Drawing.new("Line")
                        line.Color        = ESPState.Accent
                        line.Thickness    = 2
                        line.Transparency = 1
                        line.ZIndex       = 3
                        table.insert(esp.Lines, line)
                    end
                end
                local isR15 = hum and hum.RigType == Enum.HumanoidRigType.R15 or character:FindFirstChild("UpperTorso") ~= nil
                local links = isR15 and R15_LINKS or R6_LINKS
                for i, line in ipairs(esp.Lines) do
                    local link = links[i]
                    if link then
                        local part1 = character:FindFirstChild(link[1])
                        local part2 = character:FindFirstChild(link[2])
                        if part1 and part2 then
                            local pos1, vis1 = workspace.CurrentCamera:WorldToViewportPoint(part1.Position)
                            local pos2, vis2 = workspace.CurrentCamera:WorldToViewportPoint(part2.Position)
                            if vis1 or vis2 then
                                line.From    = Vector2.new(pos1.X, pos1.Y)
                                line.To      = Vector2.new(pos2.X, pos2.Y)
                                line.Visible = true
                            else line.Visible = false end
                        else line.Visible = false end
                    else line.Visible = false end
                end
            else
                for _, line in ipairs(esp.Lines) do pcall(function() line:Remove() end) end
                esp.Lines = {}
            end
        else
            esp.Tracer.Visible = false
            for _, line in ipairs(esp.Lines) do line.Visible = false end
        end
    else
        esp.Tracer.Visible = false
        esp.Highlight.Enabled = false
        esp.Highlight.Adornee = nil
        for _, line in ipairs(esp.Lines) do line.Visible = false end
    end
end

local function createESPForPlayer(plr)
    if plr == Player then return end
    removeESP(plr)
    local esp = createESP()
    espActivePlayers[plr] = { esp = esp }
end

local function enableESP()
    if espEnabled then return end
    espEnabled = true
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= Player then task.spawn(function() createESPForPlayer(plr) end) end
    end
    espActivePlayers._joinConn = Players.PlayerAdded:Connect(function(plr)
        if not espEnabled or plr == Player then return end
        task.wait(0.3)
        createESPForPlayer(plr)
    end)
    espActivePlayers._leaveConn = Players.PlayerRemoving:Connect(function(plr) removeESP(plr) end)
    espActivePlayers._renderConn = RunService.RenderStepped:Connect(function()
        for plr, data in pairs(espActivePlayers) do
            if typeof(plr) == "Instance" and plr:IsA("Player") and type(data) == "table" and data.esp then
                pcall(function() updateESP(plr, data.esp) end)
            end
        end
    end)
end

local function disableESP()
    espEnabled = false
    if espActivePlayers._joinConn then pcall(function() espActivePlayers._joinConn:Disconnect() end); espActivePlayers._joinConn = nil end
    if espActivePlayers._leaveConn then pcall(function() espActivePlayers._leaveConn:Disconnect() end); espActivePlayers._leaveConn = nil end
    if espActivePlayers._renderConn then pcall(function() espActivePlayers._renderConn:Disconnect() end); espActivePlayers._renderConn = nil end
    for plr in pairs(espActivePlayers) do
        if typeof(plr) == "Instance" and plr:IsA("Player") and plr ~= Player then removeESP(plr) end
    end
end

local function refreshESPState()
    local anyEsp = ESPState.Players or ESPState.Tracers or ESPState.Skeleton
    if anyEsp then
        if not espEnabled then enableESP() end
    else
        if espEnabled then disableESP() end
    end
end

local DropBrainrot = {
    Active = false,
    Mode = CONFIG.DropMode,
    Connections = {},
}

local function startDropV1()
    if DropBrainrot.Active then return end
    local char = Player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    DropBrainrot.Active = true
    local t0 = tick()
    local dc = RunService.Heartbeat:Connect(function()
        local r = char and char:FindFirstChild("HumanoidRootPart")
        if not r then dc:Disconnect(); DropBrainrot.Active = false; return end

        if tick() - t0 >= 0.2 then
            dc:Disconnect()
            local rp = RaycastParams.new()
            rp.FilterDescendantsInstances = {char}
            rp.FilterType = Enum.RaycastFilterType.Exclude
            local rr = workspace:Raycast(r.Position, Vector3.new(0, -2000, 0), rp)
            if rr then
                local hum2 = char:FindFirstChildOfClass("Humanoid")
                local off  = (hum2 and hum2.HipHeight or 2) + (r.Size.Y / 2)
                r.CFrame   = CFrame.new(r.Position.X, rr.Position.Y + off, r.Position.Z)
                r.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            end
            DropBrainrot.Active = false
            return
        end

        r.AssemblyLinearVelocity = Vector3.new(
            r.AssemblyLinearVelocity.X, 150, r.AssemblyLinearVelocity.Z
        )
    end)
    table.insert(DropBrainrot.Connections, dc)
end

local function startDropV2()
    if DropBrainrot.Active then return end
    local char = Player.Character
    if not char then return end
    if not char:FindFirstChild("HumanoidRootPart") then return end
    DropBrainrot.Active = true

    local colConn = RunService.Stepped:Connect(function()
        if not DropBrainrot.Active then return end
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= Player and plr.Character then
                for _, part in ipairs(plr.Character:GetChildren()) do
                    if part:IsA("BasePart") then part.CanCollide = false end
                end
            end
        end
    end)
    table.insert(DropBrainrot.Connections, colConn)

    local flingThread = coroutine.create(function()
        while DropBrainrot.Active do
            RunService.Heartbeat:Wait()
            local c = Player.Character
            local r = c and c:FindFirstChild("HumanoidRootPart")
            if not r then break end
            local vel = r.Velocity
            r.Velocity = vel * 10000 + Vector3.new(0, 10000, 0)
            RunService.RenderStepped:Wait()
            if r and r.Parent then r.Velocity = vel end
            RunService.Stepped:Wait()
            if r and r.Parent then r.Velocity = vel + Vector3.new(0, 0.1, 0) end
        end
    end)
    table.insert(DropBrainrot.Connections, flingThread)
    coroutine.resume(flingThread)

    task.delay(0.1, function()
        DropBrainrot.Active = false
        for _, c in ipairs(DropBrainrot.Connections) do
            if typeof(c) == "RBXScriptConnection" then c:Disconnect()
            elseif type(c) == "thread" then pcall(coroutine.close, c) end
        end
        DropBrainrot.Connections = {}
    end)
end

local function runDrop()
    if DropBrainrot.Mode == "v2" then
        startDropV2()
    else
        startDropV1()
    end
end

    function UI.speedRow(page, labelText, order, value, configKey, callback, hasBind)
        local row = UI.makeRow(page, order, labelText)

        local bindBtn = nil

            bindBtn.MouseButton1Click:Connect(function()
                if listeningName == configKey then
                    listeningName = nil
                    refreshBindButton(configKey)

                end

                bindBtn.Text = "..."
                bindBtn.BackgroundColor3 = PALETTE.bgInput
            end)
        end

        box:GetPropertyChangedSignal("Text"):Connect(function()
            local v = tonumber(box.Text)

                callback(v)

        box.FocusLost:Connect(function()
            local v = tonumber(box.Text)
            if v then callback(v) end

                tween(tabBtn, {BackgroundColor3 = tName == name and PALETTE.accent or PALETTE.bgRow, BackgroundTransparency = tName == name and 0 or 0.5})

                if tabStroke then
                    tween(tabStroke, {Color = tName == name and PALETTE.accentBright or PALETTE.strokeSoft, Transparency = tName == name and 0 or 0.6}, 0.15)
                end
            end
        end
    end

        end
    end)
